# 캠핑·야영지 사이트 — 개발 기획서 v1 (Claude Code 핸드오프)

> **목적**: Claude Code가 픽업해 즉시 실행할 수 있는 *executable handoff*.
> v3·v3.1·디자인 프롬프트의 결정을 *코드 가능한 명세*로 변환.

---

## 0. 사용 가이드

### 0-1. 본 문서의 위치
- **비즈니스/SEO/컴플라이언스 결정** → `야영지가이드_v3_본기획서.md` (왜)
- **블로그 섹션** → `야영지가이드_v3.1_블로그기획.md`
- **시각 디자인** → `야영지가이드_디자인프롬프트_v1.md`
- **본 문서** → 무엇을·어떻게 (Phase 1~5 실행 명세)

### 0-2. Claude Code 작업 방법

1. 본 문서 + v3 + v3.1 + 디자인 프롬프트 첨부
2. Phase 1부터 순차 실행 — 한 번에 Phase 1~5 다 하지 말 것
3. 각 Phase 끝에 *사용자 승인* 받고 다음 Phase
4. 멀티 인스턴스 가능: Phase 2(수집)와 Phase 3(AI 보강 파이프라인 설계)는 병렬
5. 의문 시: v3 본 기획서가 source of truth

### 0-3. 코딩 원칙 (v3 섹션 10 + 본 문서)
- TypeScript strict
- Drizzle ORM (raw SQL 최소)
- 에러 핸들링 철저 (특히 외부 API)
- 모듈화 (lib/ 기능별 분리)
- 체크포인트 패턴 (대량 작업)
- Vercel 함수 10초/60초 고려 (장시간은 GH Actions)
- 한국어 주석 OK, 변수/함수명 영어
- RSC 우선, 'use client'는 인터랙션만

## 3. DB 스키마 (Drizzle TypeScript)

`lib/db/schema.ts`:

```typescript
import { sqliteTable, text, integer, real, primaryKey, index, uniqueIndex } from 'drizzle-orm/sqlite-core';
import { sql } from 'drizzle-orm';

// 1. data_sources — API 소스 관리
export const dataSources = sqliteTable('data_sources', {
  id: text('id').primaryKey(),
  name: text('name').notNull(),
  url: text('url'),
  license: text('license'),                 // 공공누리 제1~4유형
  lastFetchedAt: integer('last_fetched_at', { mode: 'timestamp' }),
  notes: text('notes'),
});

// 2. campsites — 야영지 raw + enrichment
export const campsites = sqliteTable('campsites', {
  id: text('id').primaryKey(),              // gocamping ID 또는 합성
  sourceId: text('source_id').notNull().references(() => dataSources.id),
  externalId: text('external_id'),          // 원본 ID
  name: text('name').notNull(),
  nameNormalized: text('name_normalized'),  // dedup용
  sido: text('sido').notNull(),
  gungu: text('gungu'),
  address: text('address'),
  lat: real('lat'),
  lng: real('lng'),
  type: text('type'),                       // 'campsite' | 'hugyeso' | 'park'
  operator: text('operator'),               // 운영기관
  operatorType: text('operator_type'),      // 'public_central' | 'public_local' | 'private' | 'unknown'
  price1Night: integer('price_1night'),     // KRW
  facilities: text('facilities', { mode: 'json' }), // {electric, hotwater, toilet, ...}
  photos: text('photos', { mode: 'json' }), // [{url, source, license}]
  contact: text('contact'),
  reservationUrl: text('reservation_url'),
  rawData: text('raw_data', { mode: 'json' }), // 전체 원본
  
  // 자동 라벨
  isPublic: integer('is_public', { mode: 'boolean' }).default(false),
  isFree: integer('is_free', { mode: 'boolean' }).default(false),
  isCheap: integer('is_cheap', { mode: 'boolean' }).default(false),  // <= 30000
  isChabak: integer('is_chabak', { mode: 'boolean' }).default(false),
  chabakTrustLevel: text('chabak_trust_level'), // 'confirmed'|'estimated'|'reported'|'incident'|null
  chabakSource: text('chabak_source'),      // 출처 URL
  chabakSourceDate: text('chabak_source_date'),
  
  fitScore: integer('fit_score').default(0),
  
  // 합성 데이터
  uniquePoints: text('unique_points', { mode: 'json' }), // [{type, value, note}]
  distanceFromCities: text('distance_from_cities', { mode: 'json' }), // {seoul: {km, mins}, busan: ...}
  nearbyTourSpots: text('nearby_tour_spots', { mode: 'json' }),
  regionAvgComparison: text('region_avg_comparison', { mode: 'json' }),
  safetyScore: integer('safety_score'),
  
  createdAt: integer('created_at', { mode: 'timestamp' }).default(sql`(unixepoch())`),
  updatedAt: integer('updated_at', { mode: 'timestamp' }).default(sql`(unixepoch())`),
}, (t) => ({
  sidoIdx: index('campsites_sido_idx').on(t.sido),
  gunguIdx: index('campsites_gungu_idx').on(t.sido, t.gungu),
  fitScoreIdx: index('campsites_fitscore_idx').on(t.fitScore),
  isPublicIdx: index('campsites_public_idx').on(t.isPublic),
  isChabakIdx: index('campsites_chabak_idx').on(t.isChabak),
  nameNormIdx: index('campsites_namenorm_idx').on(t.nameNormalized),
}));

// 3. dedup_review — 수동 검토 큐
export const dedupReview = sqliteTable('dedup_review', {
  id: integer('id').primaryKey({ autoIncrement: true }),
  candidateAId: text('candidate_a_id').notNull().references(() => campsites.id),
  candidateBId: text('candidate_b_id').notNull().references(() => campsites.id),
  similarity: real('similarity').notNull(),
  status: text('status').default('pending'), // 'pending'|'merged'|'distinct'|'skipped'
  resolvedAt: integer('resolved_at', { mode: 'timestamp' }),
  resolvedBy: text('resolved_by'),
});

// 4. pages — 프로그래매틱 페이지 (L1·L2·L3 통합)
export const pages = sqliteTable('pages', {
  id: text('id').primaryKey(),
  type: text('type').notNull(),             // 'campsite_l1' | 'theme_l2' | 'theme_region_l2' | 'season_l2' | 'region_l2' | 'matching_prefilled' | 'evergreen_l3' | 'guide_l2'
  slug: text('slug').notNull(),
  url: text('url').notNull(),               // canonical
  campsiteId: text('campsite_id').references(() => campsites.id), // L1만
  theme: text('theme'),                     // L2만
  sido: text('sido'),
  gungu: text('gungu'),
  season: text('season'),                   // L2 시즌
  
  title: text('title').notNull(),
  metaDescription: text('meta_description'),
  
  commentary: text('commentary'),           // AI 코멘터리
  faqs: text('faqs', { mode: 'json' }),     // [{q, a}]
  activeSections: text('active_sections', { mode: 'json' }), // 조건부 섹션
  internalLinks: text('internal_links', { mode: 'json' }),
  
  // 품질 게이트
  qualityScore: real('quality_score').default(0),
  gatePassed: integer('gate_passed', { mode: 'boolean' }).default(false),
  gateResults: text('gate_results', { mode: 'json' }), // {gate1: true, ...}
  cosineSimilarity: real('cosine_similarity'),
  uniquePointsCount: integer('unique_points_count').default(0),
  
  // 상태
  status: text('status').notNull().default('draft'), // 'draft'|'enriched'|'quality_passed'|'published'|'demoted'|'rejected'
  publishedAt: integer('published_at', { mode: 'timestamp' }),
  
  // 메타데이터
  datePublished: text('date_published'),    // ISO date
  dateModified: text('date_modified'),
  lastReviewedAt: integer('last_reviewed_at', { mode: 'timestamp' }),
  persona: text('persona'),                 // 'saver' | 'traveler' | 'analyst'
  authorLabel: text('author_label'),
  
  // ISR
  revalidateSeconds: integer('revalidate_seconds').default(604800), // 기본 7일
  
  createdAt: integer('created_at', { mode: 'timestamp' }).default(sql`(unixepoch())`),
  updatedAt: integer('updated_at', { mode: 'timestamp' }).default(sql`(unixepoch())`),
}, (t) => ({
  urlIdx: uniqueIndex('pages_url_idx').on(t.url),
  typeStatusIdx: index('pages_type_status_idx').on(t.type, t.status),
  themeRegionIdx: index('pages_theme_region_idx').on(t.theme, t.sido),
  publishedIdx: index('pages_published_idx').on(t.status, t.publishedAt),
}));

// 5. blog_posts — 블로그 글
export const blogPosts = sqliteTable('blog_posts', {
  id: text('id').primaryKey(),
  slug: text('slug').notNull().unique(),
  title: text('title').notNull(),
  category: text('category').notNull(),     // 14 클러스터
  tags: text('tags', { mode: 'json' }),
  
  bodyMarkdown: text('body_markdown'),
  metaDescription: text('meta_description'),
  heroImage: text('hero_image'),
  
  faqs: text('faqs', { mode: 'json' }),
  internalLinks: text('internal_links', { mode: 'json' }),
  externalSources: text('external_sources', { mode: 'json' }),
  
  // SEO
  primaryKeyword: text('primary_keyword'),
  secondaryKeywords: text('secondary_keywords', { mode: 'json' }),
  
  // 페르소나
  persona: text('persona'),
```

```typescript
import { env } from '@/lib/utils/env';

const responseSchema = z.object({
  response: z.object({
    header: z.object({
      resultCode: z.string(),
      resultMsg: z.string(),
    }),
    body: z.object({
      items: z.object({
        item: z.array(z.unknown()),
      }).optional(),
      totalCount: z.number(),
      pageNo: z.number(),
      numOfRows: z.number(),
    }),
  }),
});

export class GoCampingClient {
  private base = 'https://apis.data.go.kr/B551011/GoCamping';
  private key = env.GOCAMPING_API_KEY;

  async fetchPage(pageNo: number, numOfRows = 100) {
    const url = new URL(`${this.base}/basedList`);
    url.searchParams.set('serviceKey', this.key);
    url.searchParams.set('numOfRows', String(numOfRows));
    url.searchParams.set('pageNo', String(pageNo));
    url.searchParams.set('MobileOS', 'ETC');
    url.searchParams.set('MobileApp', 'CampgroundSite');
    url.searchParams.set('_type', 'json');

    for (let attempt = 0; attempt < 3; attempt++) {
      try {
        const res = await fetch(url, { 
          next: { revalidate: 86400 },
          signal: AbortSignal.timeout(30000),
        });
        if (!res.ok) throw new Error(`HTTP ${res.status}`);
        const json = await res.json();
        const parsed = responseSchema.parse(json);
        if (parsed.response.header.resultCode !== '0000') {
          throw new Error(`API ${parsed.response.header.resultMsg}`);
        }
        return parsed.response.body;
      } catch (err) {
        if (attempt === 2) throw err;
        await new Promise(r => setTimeout(r, 1000 * Math.pow(2, attempt)));
      }
    }
    throw new Error('unreachable');
  }
}
```

```typescript
export function calcFitScore(c: Campsite): number {
  let score = 0;
  if (c.isPublic) score += 30;
  if (c.isFree || c.isCheap) score += 25;
  if (c.chabakTrustLevel === 'confirmed') score += 20;
  else if (c.chabakTrustLevel === 'estimated') score += 10;
  
  const facilityKeys = ['toilet', 'water', 'electric', 'hotwater', 'shower', 'store'];
  const facilityCount = facilityKeys.filter(k => (c.facilities as any)?.[k]).length;
  score += Math.min(15, facilityCount * 2);
  
  if (Array.isArray(c.photos) && c.photos.length >= 1) score += 10;
  
  // 인기 유료 글램핑/오토 감점
  if (c.operatorType === 'private' && c.price1Night && c.price1Night > 50000) score -= 10;
  
  return score;
}
```

```typescript
export function classifyChabakTrust(c: Campsite, sources: ChabakSource[]): { level: ChabakTrust; source?: string; date?: string } {
  const officialNotice = sources.find(s => s.type === 'official_notice' && s.campsiteId === c.id);
  if (officialNotice) return { level: 'confirmed', source: officialNotice.url, date: officialNotice.date };
  
  if (c.isPublic && !c.facilities?.sitesRigid) {
    return { level: 'estimated' };
  }
  
  const userReport = sources.find(s => s.type === 'user_report' && s.campsiteId === c.id && s.verified);
  if (userReport) return { level: 'reported', date: userReport.date };
  
  const incident = sources.find(s => s.type === 'incident' && s.campsiteId === c.id);
  if (incident) return { level: 'incident', source: incident.url, date: incident.date };
  
  return { level: null };
}
```

```typescript
import { GoogleGenerativeAI } from '@google/generative-ai';
import { env } from '@/lib/utils/env';
import { db } from '@/lib/db/client';
import { aiCache, aiBudget } from '@/lib/db/schema';
import { eq } from 'drizzle-orm';
import crypto from 'node:crypto';

const FLASH_MODEL = 'gemini-2.5-flash'; // 최신 Flash 확인 후 변경
const PRICE_INPUT_PER_1M = 0.075;       // USD, 모델 가격 확인 후 갱신
const PRICE_OUTPUT_PER_1M = 0.30;

const genAI = new GoogleGenerativeAI(env.GEMINI_API_KEY);
const model = genAI.getGenerativeModel({ model: FLASH_MODEL });

export async function generate({ prompt, promptVersion, dryRun = false }: { prompt: string; promptVersion: string; dryRun?: boolean }) {
  const inputHash = crypto.createHash('sha256').update(`${promptVersion}:${prompt}`).digest('hex');
  
  // 캐시 확인
  const cached = await db.select().from(aiCache).where(eq(aiCache.inputHash, inputHash)).get();
  if (cached) return { output: cached.output, fromCache: true, costUsd: 0 };
  
  // 일 한도 확인
  const today = new Date().toISOString().slice(0, 10);
  const todayBudget = await db.select().from(aiBudget).where(eq(aiBudget.date, today)).get();
  if (todayBudget && todayBudget.totalCostUsd! >= env.GEMINI_DAILY_BUDGET_USD) {
    throw new Error(`Daily budget exceeded: $${todayBudget.totalCostUsd} / $${env.GEMINI_DAILY_BUDGET_USD}`);
  }
  
  if (dryRun) {
    return { output: '[dry-run]', fromCache: false, costUsd: 0 };
  }
  
  // 호출
  const result = await model.generateContent(prompt);
  const output = result.response.text();
  const usage = result.response.usageMetadata;
  const inputTokens = usage?.promptTokenCount ?? 0;
  const outputTokens = usage?.candidatesTokenCount ?? 0;
  const costUsd = (inputTokens / 1_000_000) * PRICE_INPUT_PER_1M + (outputTokens / 1_000_000) * PRICE_OUTPUT_PER_1M;
  
  // 캐시 + 예산 기록
  await db.insert(aiCache).values({ inputHash, promptVersion, output, inputTokens, outputTokens, costUsd });
  await db.insert(aiBudget)
    .values({ date: today, totalCostUsd: costUsd, callCount: 1 })
    .onConflictDoUpdate({ target: aiBudget.date, set: { totalCostUsd: sql`total_cost_usd + ${costUsd}`, callCount: sql`call_count + 1` } });
  
  return { output, fromCache: false, costUsd };
}
```

```typescript
  // 3. 제목 유사도 < 30%
  results.titleUniqueness = await checkTitleUniqueness(page.title, page.id);
  if (!results.titleUniqueness) reasons.push('title similar to existing (≥ 30%)');
  
  // 4. FAQ 3+
  const faqs = page.faqs as any[] || [];
  results.faqs = faqs.length >= 3 && faqs.every(f => f.a.length >= 50);
  if (!results.faqs) reasons.push('FAQ < 3 or answers < 50자');
  
  // 5. 조건부 섹션 1+ (L1만)
  if (page.type === 'campsite_l1') {
    const sections = page.activeSections as any[] || [];
    results.conditionalSections = sections.length >= 1;
    if (!results.conditionalSections) reasons.push('no active conditional sections');
  } else {
    results.conditionalSections = true;
  }
  
  // 6. 비교 데이터 1+
  results.comparison = !!(campsite?.regionAvgComparison);
  if (!results.comparison) reasons.push('no comparison data');
  
  // 7. 실사진 1+ (L1만)
  if (page.type === 'campsite_l1') {
    const photos = campsite?.photos as any[] || [];
    results.photos = photos.length >= 1;
    if (!results.photos) reasons.push('no photo');
  } else {
    results.photos = true;
  }
  
  // 8. 코사인 유사도 (캘리브레이션된 임계)
  const threshold = await getCurrentCosineThreshold();
  results.cosine = (page.cosineSimilarity ?? 1) < threshold;
  if (!results.cosine) reasons.push(`cosine ≥ ${threshold}`);
  
  const passedCount = Object.values(results).filter(Boolean).length;
  const score = passedCount / Object.keys(results).length;
  
  return { passed: passedCount === Object.keys(results).length, results, score, failureReasons: reasons };
}
```

### 6-5. Phase 0.6 캘리브레이션 (`scripts/calibrate-cosine.ts`)

```typescript
// Python 임시 호출 또는 JS 형태소 분석 라이브러리 사용
// 베이스라인 측정 → 임계 결정 → DB 저장

async function calibrate() {
  const samples = await db.select().from(pages).where(eq(pages.status, 'enriched')).limit(30).all();
  const similarities: number[] = [];
  
  for (const p of samples) {
    const campsite = await getCampsite(p.campsiteId);
    if (!campsite) continue;
    const originalText = `${campsite.name} ${(campsite.rawData as any)?.intro || ''}`;
    const ourText = `${p.title} ${p.commentary}`;
    const sim = await cosineSimilarity(originalText, ourText);
    similarities.push(sim);
  }
  
  const mean = similarities.reduce((a, b) => a + b, 0) / similarities.length;
  const stddev = Math.sqrt(similarities.reduce((a, b) => a + (b - mean) ** 2, 0) / similarities.length);
  const threshold = mean + 2 * stddev; // 보수
  
  await db.insert(cosineCalibration).values({
    sampleSize: samples.length,
    meanSimilarity: mean,
    stddev,
    recommendedThreshold: threshold,
    notes: `mean ${mean.toFixed(3)} + 2σ ${stddev.toFixed(3)} = ${threshold.toFixed(3)}`,
  });
  
  console.log(`Calibrated threshold: ${threshold.toFixed(3)} (사용 시 page.cosine < ${threshold.toFixed(3)}이면 통과)`);
}
```

### 7-3. 매칭 클라이언트 (`lib/matching/client-matcher.ts`)

```typescript
'use client';

import { useState, useEffect } from 'react';

let cache: CampsiteMatchData[] | null = null;
async function loadData() {
  if (cache) return cache;
  const res = await fetch('/matching-data.json');
  cache = await res.json();
  return cache!;
}

function haversine(lat1: number, lng1: number, lat2: number, lng2: number): number {
  const R = 6371;
  const dLat = (lat2 - lat1) * Math.PI / 180;
  const dLng = (lng2 - lng1) * Math.PI / 180;
  const a = Math.sin(dLat/2)**2 + Math.cos(lat1*Math.PI/180) * Math.cos(lat2*Math.PI/180) * Math.sin(dLng/2)**2;
  return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1-a));
}

export async function runMatching(input: MatchInput): Promise<CampsiteMatchData[]> {
  const all = await loadData();
  
  const filtered = all.filter(c => {
    if (input.themes?.length && !input.themes.every(t => c.themes.includes(t))) return false;
    if (input.budget === 'free' && c.price1Night > 0) return false;
    if (input.budget === '10000' && c.price1Night > 10000) return false;
    if (input.budget === '30000' && c.price1Night > 30000) return false;
    if (input.legalityFilter === 'confirmed_only' && c.chabakTrustLevel !== 'confirmed') return false;
    return true;
  });
  
  if (input.from) {
    const origin = CITY_COORDS[input.from];
    if (origin) {
      const radius = input.radius ?? 100;
      return filtered
        .map(c => ({ c, dist: haversine(origin.lat, origin.lng, c.lat, c.lng) }))
        .filter(x => x.dist <= radius)
        .sort((a, b) => a.dist - b.dist)
        .slice(0, 10)
        .map(x => x.c);
    }
  }
  
  return filtered.sort((a, b) => b.fitScore - a.fitScore).slice(0, 10);
}
```
